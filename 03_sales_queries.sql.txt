-- 1. View all customers
SELECT *
FROM customers;

-- 2. View all services
SELECT service_name, category, base_price
FROM services
ORDER BY base_price DESC;

-- 3. Customer order report
SELECT
    o.order_id,
    c.customer_name,
    s.service_name,
    o.order_date,
    o.quantity,
    o.unit_price,
    o.quantity * o.unit_price AS total_amount
FROM orders o
JOIN customers c
    ON o.customer_id = c.customer_id
JOIN services s
    ON o.service_id = s.service_id
ORDER BY o.order_id;

-- 4. Total sales revenue
SELECT SUM(quantity * unit_price) AS total_sales
FROM orders;

-- 5. Sales by customer
SELECT
    c.customer_name,
    SUM(o.quantity * o.unit_price) AS total_spent
FROM customers c
JOIN orders o
    ON c.customer_id = o.customer_id
GROUP BY c.customer_id, c.customer_name
ORDER BY total_spent DESC;

-- 6. Sales by service
SELECT
    s.service_name,
    SUM(o.quantity * o.unit_price) AS total_sales
FROM services s
JOIN orders o
    ON s.service_id = o.service_id
GROUP BY s.service_id, s.service_name
ORDER BY total_sales DESC;
